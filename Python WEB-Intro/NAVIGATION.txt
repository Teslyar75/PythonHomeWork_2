│ BASIC SYNTAX                                                             │
├──────────────────────────────────────────────────────────────────────────┤
│                                                                          │
│  # Creation                                                             │
│  my_list = [1, 2, 3]                                                    │
│  my_tuple = (1, 2, 3)                                                   │
│  my_set = {1, 2, 3}                                                     │
│  my_dict = {"a": 1, "b": 2}                                             │
│                                                                          │
│  # Comprehensions                                                       │
│  [x**2 for x in range(10)]          # List                             │
│  {x**2 for x in range(10)}          # Set                              │

│  (x**2 for x in range(10))          # Generator                        │
│                                                                          │
│  # Lambda and higher-order functions                                    │
│  lambda x: x**2                      # Anonymous function               │

│  filter(func, iterable)              # Filter elements                 │
│  sorted(iterable, key=func)          # Sort by function                │

│  # Generators                                                           │
│  def gen():                                                             │
│      yield value                     # Return and "freeze"             │

# Базове замикання
def make_multiplier(n):
    def multiplier(x):
        return x * n  # n запам'ятовується!
    return multiplier

multiply_by_5 = make_multiplier(5)
multiply_by_5(10)  # 50

# Замикання зі станом
def make_counter():
    count = 0
    def counter():
        nonlocal count  # Змінюємо змінну зовнішньої функції
        count += 1
        return count
    return counter

c = make_counter()
c()  # 1
c()  # 2
c()  # 3

# Декоратор (приклад замикання)
def logger(func):
    def wrapper(*args, **kwargs):
        print(f"Виклик {func.__name__}")
        result = func(*args, **kwargs)
        print(f"Результат: {result}")
        return result
    return wrapper

@logger
def add(a, b):
    return a + b

# Функція-генератор
def countdown(n):
    while n > 0:
        yield n  # Повертає і "заморожує" функцію
        n -= 1

gen = countdown(5)
next(gen)  # 5
next(gen)  # 4

# Нескінченний генератор
def infinite_counter():
    n = 0
    while True:
        yield n
        n += 1

# Генератор Фібоначчі
def fibonacci():
    a, b = 0, 1
    while True:
        yield a
        a, b = b, a + b

# Generator expression

# Використання з функціями
sum(x**2 for x in range(10))  # Сума квадратів
max(x for x in range(100) if x % 7 == 0)  # Максимальне кратне 7

# yield from (делегування)
def combined():
    yield from range(3)  # 0, 1, 2
    yield from range(3, 6)  # 3, 4, 5
